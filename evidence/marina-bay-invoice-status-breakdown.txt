SELECT
    status,
    COUNT(*) AS invoice_count,
    COUNT(*) FILTER (
        WHERE NULLIF(BTRIM(asp_message_id), '') IS NOT NULL
    ) AS with_asp_message_id,
    COUNT(*) FILTER (
        WHERE NULLIF(BTRIM(asp_message_id), '') IS NULL
    ) AS without_asp_message_id
FROM invoices
WHERE seller_id = 3
